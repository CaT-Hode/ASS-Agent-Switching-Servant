cmake_minimum_required(VERSION 3.24)
project(ASSRequestGlass LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
set(DESKTOP_GLASS_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
add_subdirectory(desktop-glass)
add_library(RequestGlass MODULE request-glass.cpp delay-load.cpp)
target_compile_features(RequestGlass PRIVATE cxx_std_20)
target_compile_definitions(RequestGlass PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX NAPI_VERSION=8)
target_include_directories(RequestGlass PRIVATE "${NODE_INCLUDE}")
target_link_libraries(RequestGlass PRIVATE DesktopGlass "${NODE_LIBRARY}" user32 dwmapi delayimp)
target_link_options(RequestGlass PRIVATE "/DELAYLOAD:node.exe")
target_compile_options(RequestGlass PRIVATE /W4 /EHsc /utf-8)
set_target_properties(RequestGlass PROPERTIES PREFIX "" SUFFIX ".node")
if(BUILD_TESTING)
  add_library(RequestGlassReview MODULE request-glass.cpp delay-load.cpp)
  target_compile_features(RequestGlassReview PRIVATE cxx_std_20)
  target_compile_definitions(RequestGlassReview PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX NAPI_VERSION=8 ASS_GLASS_REVIEW)
  target_include_directories(RequestGlassReview PRIVATE "${NODE_INCLUDE}")
  target_link_libraries(RequestGlassReview PRIVATE DesktopGlass "${NODE_LIBRARY}" user32 dwmapi delayimp)
  target_link_options(RequestGlassReview PRIVATE "/DELAYLOAD:node.exe")
  target_compile_options(RequestGlassReview PRIVATE /W4 /EHsc /utf-8)
  set_target_properties(RequestGlassReview PROPERTIES PREFIX "" SUFFIX ".node")
endif()
