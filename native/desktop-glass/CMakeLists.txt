cmake_minimum_required(VERSION 3.24)
project(DesktopGlass LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "DesktopGlass requires Windows and the Microsoft C++ toolchain")
endif()

add_library(DesktopGlass SHARED DesktopGlass.cpp)
target_compile_features(DesktopGlass PUBLIC cxx_std_20)
target_compile_definitions(DesktopGlass PRIVATE DESKTOP_GLASS_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(DesktopGlass PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(DesktopGlass PRIVATE d3d11 dxgi d3dcompiler dcomp user32 windowsapp ole32)
target_compile_options(DesktopGlass PRIVATE /W4 /EHsc /utf-8)

# The C++/WinRT headers ship in the Windows SDK; no NuGet or project checkout.
if(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION)
  get_filename_component(_sdk "[HKEY_LOCAL_MACHINE\\SOFTWARE\\Microsoft\\Windows Kits\\Installed Roots;KitsRoot10]" ABSOLUTE)
  target_include_directories(DesktopGlass PRIVATE "${_sdk}/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt")
endif()

option(DESKTOP_GLASS_BUILD_EXAMPLE "Build the opt-in interactive Win32 example" ON)
if(DESKTOP_GLASS_BUILD_EXAMPLE)
  add_executable(desktop-glass-example WIN32 examples/minimal.cpp)
  target_compile_features(desktop-glass-example PRIVATE cxx_std_20)
  target_compile_definitions(desktop-glass-example PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_link_libraries(desktop-glass-example PRIVATE DesktopGlass user32 gdi32 wtsapi32)
  target_compile_options(desktop-glass-example PRIVATE /W4 /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(desktop-glass-smoke tests/shader_smoke.cpp)
  target_compile_features(desktop-glass-smoke PRIVATE cxx_std_20)
  target_compile_definitions(desktop-glass-smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(desktop-glass-smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(desktop-glass-smoke PRIVATE DesktopGlass d3d11 d3dcompiler dxgi)
  target_compile_options(desktop-glass-smoke PRIVATE /W4 /utf-8)
  add_test(NAME desktop-glass-synthetic-gpu COMMAND desktop-glass-smoke)
endif()
